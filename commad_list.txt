git clone https://github.com/kamnii/GitTest.git    (скачивает удаленный репозиторий на ваш компьютер)
git init    (создает новый Git-репозиторий в текущей папке)
git add .   (Добавляет все измененные/новые файлы в "индекс" (staging area))
git commit -m "commit_name"    (сохраняет изменения в истории репозитория)
git remote add origin https://github.com/kamnii/GitTest.git    (привязывает локальный репозиторий к удаленному)
git push -u origin master   (отправляет коммиты на GitHub)

git branch    (показывает список всех локальных веток, звездочкой * отмечена текущая ветка)
git branch название_новой_ветки    (создает новую ветку с указанным названием, но НЕ переключается на нее)
git branch -m новое_название_ветки    (переименовывает текущую ветку в новое название)
git branch -d название_ветки    (удаляет ветку, только если она уже слита с другой веткой; для принудительного удаления используется -D)
git checkout название_ветки    (переключается на указанную существующую ветку)
git checkout -b название_новой_ветки    (создает новую ветку и сразу переключается на нее)
git checkout хеш_коммита    (переключается на конкретный коммит по его хешу — состояние "detached HEAD")

git status    (показывает состояние файлов)
git status --short    (сокращенный формат статуса (компактный))
git log    (показывает историю коммитов (автор, дата, хеш, сообщение))
git log --oneline   (история коммитов одной строкой — хеш (7 символов) + сообщение)

git checkout -- .   (откатывает все изменения в файлах до последнего коммита)
git reset .    (убирает файлы из staging (индекса), но сохраняет изменения в файлах)
git reset --hard HEAD^1    (жесткий откат — удаляет последний коммит и все его изменения)
git reset --soft HEAD^1    (мягкий откат - удаляет последний коммит, но изменения останутся готовыми к коммиту)

git config --global alias.сокращение "полная команда"   (создает alias команды)
git config --global --get-regexp alias    (выводит список созданных алиасов)
git config --global --unset alias.название    (удаляет созданный алиас)
